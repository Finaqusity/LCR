cmake_minimum_required(VERSION 3.20)
project(lcr_next_core LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(lcr_core
  src/lcr_core.c src/lcr_frontend.c src/lcr_measure.c src/lcr_session.c
  src/lcr_cal_store.c src/lcr_journal.c src/lcr_ui.c src/lcr_app.c)
target_include_directories(lcr_core PUBLIC include)
if(MSVC)
  target_compile_options(lcr_core PRIVATE /W4 /WX)
elseif(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(lcr_core PRIVATE -Wall -Wextra -Werror)
endif()
if(NOT MSVC)
  target_link_libraries(lcr_core PUBLIC m)
endif()
option(LCR_BUILD_TESTS "Build host-executable numerical tests" ON)
if(LCR_BUILD_TESTS)
  enable_testing()
  foreach(suite core frontend measure session cal_store journal ui app settling regression)
    add_executable(test_${suite} tests/test_${suite}.c)
    target_link_libraries(test_${suite} PRIVATE lcr_core)
    # Assertions must remain active in Release too.
    if(MSVC)
      target_compile_options(test_${suite} PRIVATE /UNDEBUG /W4 /WX)
    else()
      target_compile_options(test_${suite} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
    add_test(NAME lcr_${suite} COMMAND test_${suite})
  endforeach()
endif()
